frontend/
 ├─ src/
 │  ├─ assets/              # Static files (images, icons, fonts)
 │  │  └─ logo.svg
 │  ├─ components/          # Global, reusable UI components (Atomic design)
 │  │  ├─ Button.tsx
 │  │  └─ Input.tsx
 │  ├─ features/            # Business domains (The "core" of the app)
 │  │  ├─ auth/
 │  │  │  ├─ AuthForm.tsx
 │  │  │  └─ authService.ts
 │  │  ├─ dashboard/
 │  │  │  ├─ Dashboard.tsx
 │  │  │  └─ SummaryCard.tsx
 │  │  ├─ transactions/
 │  │  │  ├─ TransactionList.tsx
 │  │  │  └─ transactionService.ts
 │  │  └─ budgets/
 │  │     ├─ BudgetList.tsx
 │  │     └─ budgetService.ts
 │  ├─ hooks/               # Global custom React hooks
 │  │  └─ useLocalStorage.ts
 │  ├─ store/               # Global state management (Zustand/Redux)
 │  │  └─ useAuthStore.ts
 │  ├─ styles/              # Global CSS/SCSS or Theme definitions
 │  │  └─ globals.css
 │  ├─ utils/               # Shared helper functions (formatters, validators)
 │  │  └─ formatCurrency.ts
 │  ├─ App.tsx              # Main application component & Routing
 │  ├─ main.tsx             # Entry point (ReactDOM.render)
 │  └─ index.tsx            # (Optional) legacy entry or type definitions
 ├─ public/                 # Static assets that don't need bundling
 ├─ package.json
 ├─ tsconfig.json
 └─ .env                    # Frontend environment variables (VITE_API_URL, etc.)
